import React, { useState } from 'react';
import {
  BookMarked,
  Plus,
  Filter,
  CheckCircle2,
  AlertTriangle,
  ArrowUpRight,
  ArrowDownRight,
  BarChart2,
  Calendar,
  Sparkles,
  Info,
  Check,
  X,
  Clock,
  Search,
} from 'lucide-react';
import { JournalTrade, TradingConstitution, PerformanceStats } from '../types';
import { fetchTradeReviewWithAI } from '../services/geminiService';

interface JournalViewProps {
  trades: JournalTrade[];
  stats: PerformanceStats;
  constitution: TradingConstitution;
  onAddTrade: (trade: JournalTrade) => void;
  onNavigateToAnalytics: () => void;
}

export const JournalView: React.FC<JournalViewProps> = ({
  trades,
  stats,
  constitution,
  onAddTrade,
  onNavigateToAnalytics,
}) => {
  const [filterType, setFilterType] = useState<'all' | 'winning' | 'losing' | 'violations'>('all');
  const [searchQuery, setSearchQuery] = useState('');
  const [selectedReviewTrade, setSelectedReviewTrade] = useState<JournalTrade | null>(null);
  const [isAiReviewLoading, setIsAiReviewLoading] = useState(false);
  const [aiReviewText, setAiReviewText] = useState<string | null>(null);
  const [isAddModalOpen, setIsAddModalOpen] = useState(false);

  // New Trade Form State
  const [newTrade, setNewTrade] = useState<Partial<JournalTrade>>({
    instrument: 'NIFTY 24850 CE',
    segment: 'Index Options',
    direction: 'BUY',
    strategy: constitution.approvedStrategies[0] || 'Pullback to 20 EMA / VWAP',
    entryPrice: 110,
    exitPrice: 155,
    stopLoss: 90,
    target: 150,
    quantity: 50,
    pnl: 2250,
    realizedR: 2.25,
    setupQuality: 85,
    session: 'Morning Trend (09:15-11:30)',
    marketCondition: 'Bullish Trend with Rising Breadth',
    ruleCompliance: true,
    ruleViolations: [],
    executionRating: 'Flawless',
    notes: '',
  });

  const filteredTrades = trades.filter((tr) => {
    const matchesSearch =
      tr.instrument.toLowerCase().includes(searchQuery.toLowerCase()) ||
      tr.strategy.toLowerCase().includes(searchQuery.toLowerCase());
    if (!matchesSearch) return false;

    if (filterType === 'winning') return tr.pnl > 0;
    if (filterType === 'losing') return tr.pnl < 0;
    if (filterType === 'violations') return !tr.ruleCompliance || tr.ruleViolations.length > 0;
    return true;
  });

  const handleOpenReview = async (trade: JournalTrade) => {
    setSelectedReviewTrade(trade);
    setAiReviewText(trade.postTradeReview || null);
    if (!trade.postTradeReview) {
      setIsAiReviewLoading(true);
      const res = await fetchTradeReviewWithAI(trade);
      if (res.success && res.review) {
        setAiReviewText(res.review);
      }
      setIsAiReviewLoading(false);
    }
  };

  const handleSaveNewTrade = (e: React.FormEvent) => {
    e.preventDefault();
    const plannedRisk = Math.abs((newTrade.entryPrice || 0) - (newTrade.stopLoss || 0)) * (newTrade.quantity || 1);
    const realizedPnL = (newTrade.pnl !== undefined ? newTrade.pnl : ((newTrade.exitPrice || 0) - (newTrade.entryPrice || 0)) * (newTrade.quantity || 1) * (newTrade.direction === 'BUY' ? 1 : -1));
    const realizedR = plannedRisk > 0 ? Number((realizedPnL / plannedRisk).toFixed(2)) : 0;

    const created: JournalTrade = {
      id: `tr-${Date.now()}`,
      date: new Date().toISOString().split('T')[0],
      time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }) + ' IST',
      instrument: newTrade.instrument || 'NIFTY',
      segment: newTrade.segment || 'Index Options',
      direction: newTrade.direction || 'BUY',
      strategy: newTrade.strategy || constitution.approvedStrategies[0],
      entryPrice: Number(newTrade.entryPrice) || 0,
      exitPrice: Number(newTrade.exitPrice) || 0,
      stopLoss: Number(newTrade.stopLoss) || 0,
      target: Number(newTrade.target) || 0,
      quantity: Number(newTrade.quantity) || 1,
      pnl: realizedPnL,
      realizedR: realizedR,
      setupQuality: Number(newTrade.setupQuality) || 80,
      session: newTrade.session || 'Morning Trend (09:15-11:30)',
      marketCondition: newTrade.marketCondition || 'Trending',
      ruleCompliance: newTrade.ruleCompliance !== undefined ? newTrade.ruleCompliance : true,
      ruleViolations: newTrade.ruleViolations || [],
      executionRating: newTrade.executionRating || 'Good',
      notes: newTrade.notes || 'Trade logged via technical journal.',
    };

    onAddTrade(created);
    setIsAddModalOpen(false);
  };

  return (
    <div className="space-y-4 pb-24 max-w-7xl mx-auto px-3 sm:px-4 lg:px-6 animate-fadeIn">
      {/* Header & Metrics Overview */}
      <div className="bg-white p-4 sm:p-5 rounded-2xl border border-slate-200 shadow-2xs space-y-4">
        <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3">
          <div>
            <div className="flex items-center gap-2">
              <div className="w-8 h-8 rounded-xl bg-indigo-50 border border-indigo-200 text-indigo-700 flex items-center justify-center">
                <BookMarked className="w-4 h-4" />
              </div>
              <h1 className="text-lg sm:text-xl font-bold text-slate-900">Technical Trading Journal</h1>
            </div>
            <p className="text-xs text-slate-500 mt-1">
              Measurable trading behavior, execution fidelity, and process vs outcome tracking.
            </p>
          </div>

          <div className="flex items-center gap-2 w-full sm:w-auto">
            <button
              onClick={() => setIsAddModalOpen(true)}
              className="flex-1 sm:flex-initial px-3.5 py-2 rounded-xl text-xs font-bold bg-indigo-600 text-white hover:bg-indigo-700 transition-all shadow-xs flex items-center justify-center gap-1.5"
            >
              <Plus className="w-4 h-4" />
              <span>Log Completed Trade</span>
            </button>
            <button
              onClick={onNavigateToAnalytics}
              className="px-3 py-2 rounded-xl text-xs font-semibold bg-slate-100 text-slate-700 hover:bg-slate-200 transition-all flex items-center gap-1"
            >
              <BarChart2 className="w-4 h-4 text-slate-600" />
              <span className="hidden sm:inline">Analytics</span>
            </button>
          </div>
        </div>

        {/* 4 Quantitative Journal Metrics */}
        <div className="grid grid-cols-2 sm:grid-cols-4 gap-2.5 pt-2 border-t border-slate-100">
          <div className="p-3 bg-slate-50 rounded-xl border border-slate-200/80">
            <span className="text-[11px] text-slate-500 block">Total Realized P&L</span>
            <span className={`text-base sm:text-lg font-bold font-mono ${(stats?.totalPnL ?? 0) >= 0 ? 'text-emerald-700' : 'text-rose-700'}`}>
              {(stats?.totalPnL ?? 0) >= 0 ? '+' : ''}₹{(stats?.totalPnL ?? 0).toLocaleString('en-IN')}
            </span>
          </div>

          <div className="p-3 bg-slate-50 rounded-xl border border-slate-200/80">
            <span className="text-[11px] text-slate-500 block">Win Rate</span>
            <span className="text-base sm:text-lg font-bold font-mono text-slate-900">
              {stats.winRate}% <span className="text-xs font-normal text-slate-500">({stats.winningTrades}W / {stats.losingTrades}L)</span>
            </span>
          </div>

          <div className="p-3 bg-slate-50 rounded-xl border border-slate-200/80">
            <span className="text-[11px] text-slate-500 block">Average Winning R</span>
            <span className="text-base sm:text-lg font-bold font-mono text-indigo-700">
              +{stats.avgWinR}R <span className="text-xs font-normal text-slate-500">(Loss -{stats.avgLossR}R)</span>
            </span>
          </div>

          <div className="p-3 bg-slate-50 rounded-xl border border-slate-200/80">
            <span className="text-[11px] text-slate-500 block">Disciplined Win Rate</span>
            <span className="text-base sm:text-lg font-bold font-mono text-emerald-700">
              {stats.disciplinedTradesWinRate}% <span className="text-[10px] text-rose-600 font-normal">({stats.undisciplinedTradesWinRate}% when breaking rules)</span>

              type="text"
              placeholder="Search instrument or strategy..."
              value={searchQuery}

            />

  Sparkles,
  Layers,
  ArrowRight,
  CheckCircle2,
  Clock,
  Flame,
  Award,
  Activity,
  Maximize2,
  DollarSign,
} from 'lucide-react';
import {
  ResponsiveContainer,
  LineChart,
  Line,
  XAxis,
  YAxis,
  Tooltip,
  CartesianGrid,
  ReferenceLine,
  Area,
  ComposedChart,
} from 'recharts';
import { PerformanceStats, BehavioralLeak, TradingConstitution, NavigationTab, JournalTrade } from '../types';
import { fetchBehavioralLeaksWithAI } from '../services/geminiService';

interface AnalyticsViewProps {
  stats: PerformanceStats;
  trades?: JournalTrade[];
  leaks: BehavioralLeak[];
  constitution: TradingConstitution;
  onApplyLeakFixToConstitution: (leak: BehavioralLeak) => void;
  onNavigate: (tab: NavigationTab) => void;
}

export const AnalyticsView: React.FC<AnalyticsViewProps> = ({
  stats,
  trades = [],
  leaks,
  constitution,
  onApplyLeakFixToConstitution,
  onNavigate,
}) => {
  const [isAiLoading, setIsAiLoading] = useState(false);
  const [aiInsight, setAiInsight] = useState<string | null>(null);
  const [chartMode, setChartMode] = useState<'pnl' | 'rMultiple'>('pnl');

  const handleGenerateAiInsight = async () => {
    setIsAiLoading(true);
    const res = await fetchBehavioralLeaksWithAI(
      trades && trades.length > 0 ? trades : [
        { symbol: 'NIFTY', pnl: -1400, ruleCompliance: false, executionRating: 'Moved SL' },
        { symbol: 'BANKNIFTY', pnl: 3200, ruleCompliance: true, executionRating: 'Flawless' },
      ],
      stats
    );
    if (res.success && res.analysis) {
      setAiInsight(res.analysis);
    }
    setIsAiLoading(false);
  };

  // Prepare chronological cumulative data points for Recharts
  const cumulativeData = useMemo(() => {
    if (!trades || trades.length === 0) {
      return [
        {
          index: 0,
          tradeNum: 0,
          label: 'Start',
          date: 'Start',
          time: '',
          instrument: 'Baseline',
          tradePnL: 0,
          realizedR: 0,
          cumulativePnL: 0,
          cumulativeR: 0,
          isCompliant: true,
          executionRating: 'Baseline',
          strategy: 'Capital Base',
        },
      ];
    }

    // Sort chronologically (oldest to newest)
    const sorted = [...trades].sort((a, b) => {
      const dateA = new Date(`${a.date} ${a.time.replace(' IST', '')}`).getTime() || new Date(a.date).getTime() || 0;
      const dateB = new Date(`${b.date} ${b.time.replace(' IST', '')}`).getTime() || new Date(b.date).getTime() || 0;
      if (dateA !== dateB) return dateA - dateB;
      return a.id.localeCompare(b.id);
    });

    let runningPnL = 0;
    let runningR = 0;

    const baselinePoint = {
      index: 0,
      tradeNum: 0,
      label: 'Start',
      date: 'Start',
      time: '09:15',
      instrument: 'Baseline',
      tradePnL: 0,
      realizedR: 0,
      cumulativePnL: 0,
      cumulativeR: 0,
      isCompliant: true,
      executionRating: 'Baseline',
      strategy: 'Starting Point',
    };

    const points = sorted.map((t, idx) => {
      runningPnL += t.pnl;
      runningR += t.realizedR;

      let dateLabel = t.date;
      try {
        const parts = t.date.split('-');
        if (parts.length === 3) {
          const monthNames = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
          const mIdx = parseInt(parts[1], 10) - 1;
          const day = parseInt(parts[2], 10);
          dateLabel = `${day} ${monthNames[mIdx] || ''}`;
        }
      } catch {
        dateLabel = t.date;
      }

      return {
        index: idx + 1,
        tradeNum: idx + 1,
        label: `T${idx + 1}`,
        date: dateLabel,
        fullDate: t.date,
        time: t.time,
        instrument: t.instrument,
        segment: t.segment,
        direction: t.direction,
        strategy: t.strategy,
        tradePnL: t.pnl,
        realizedR: Number(t.realizedR.toFixed(2)),
        cumulativePnL: runningPnL,
        cumulativeR: Number(runningR.toFixed(2)),
        isCompliant: t.ruleCompliance,
        executionRating: t.executionRating,
      };
    });

    return [baselinePoint, ...points];
  }, [trades]);

  // Derived trajectory metrics
  const trajectoryStats = useMemo(() => {
    const pnlValues = cumulativeData.map((d) => d.cumulativePnL);
    const rValues = cumulativeData.map((d) => d.cumulativeR);
    const maxPnL = Math.max(...pnlValues, 0);
    const minPnL = Math.min(...pnlValues, 0);
    const currentPnL = pnlValues[pnlValues.length - 1] ?? 0;
    const currentR = rValues[rValues.length - 1] ?? 0;
    const peakPnL = maxPnL;
    const drawdownFromPeak = peakPnL > currentPnL ? peakPnL - currentPnL : 0;

    return {
      maxPnL,
      minPnL,
      currentPnL,
      currentR,
      peakPnL,
      drawdownFromPeak,
    };
  }, [cumulativeData]);

  // Custom Chart Tooltip
  const CustomChartTooltip = ({ active, payload }: any) => {
    if (!active || !payload || !payload.length) return null;
    const data = payload[0].payload;
    if (!data) return null;

    if (data.tradeNum === 0) {
      return (
        <div className="bg-slate-900 text-white p-3 rounded-xl shadow-xl border border-slate-700 text-xs space-y-1">
          <div className="font-bold text-slate-200">Starting Equity Baseline</div>
          <div className="text-slate-400 font-mono">Cumulative P&L: ₹0 | 0.00R</div>
        </div>
      );
    }

    const isTradeWin = data.tradePnL >= 0;
    const isCumWin = data.cumulativePnL >= 0;

    return (
      <div className="bg-slate-900/95 backdrop-blur-xs text-white p-3.5 rounded-xl shadow-2xl border border-slate-700 text-xs space-y-2 min-w-[220px]">
        <div className="flex items-center justify-between border-b border-slate-800 pb-1.5">
          <span className="font-bold text-slate-200">
            Trade #{data.tradeNum}: {data.instrument}
          </span>
          <span className="text-[10px] text-slate-400 font-mono">{data.date}</span>
        </div>

        <div className="space-y-1 font-sans">
          <div className="flex justify-between items-center">
            <span className="text-slate-400">Trade Outcome:</span>
            <span className={`font-mono font-bold ${isTradeWin ? 'text-emerald-400' : 'text-rose-400'}`}>
              {isTradeWin ? '+' : ''}₹{data.tradePnL.toLocaleString('en-IN')} ({data.realizedR >= 0 ? '+' : ''}{data.realizedR}R)
            </span>
          </div>

          <div className="flex justify-between items-center">
            <span className="text-slate-400">Cumulative P&L:</span>
            <span className={`font-mono font-bold ${isCumWin ? 'text-indigo-300' : 'text-rose-300'}`}>
              {isCumWin ? '+' : ''}₹{data.cumulativePnL.toLocaleString('en-IN')}
            </span>
          </div>

          <div className="flex justify-between items-center">
            <span className="text-slate-400">Cumulative R:</span>
            <span className="font-mono font-semibold text-slate-200">
              {data.cumulativeR >= 0 ? '+' : ''}{data.cumulativeR}R
            </span>
          </div>
        </div>

        <div className="pt-1.5 border-t border-slate-800 flex items-center justify-between text-[11px]">
          <span className="text-slate-400">{data.strategy || 'Strategy'}</span>
          <span className={`font-semibold ${data.isCompliant ? 'text-emerald-400' : 'text-amber-400'}`}>
            {data.isCompliant ? '✓ Compliant' : '⚠ Rule Broken'}
          </span>
        </div>
      </div>
    );
  };

  return (
    <div className="space-y-4 pb-24 max-w-7xl mx-auto px-3 sm:px-4 lg:px-6 animate-fadeIn">
      {/* Header */}
      <div className="bg-white p-4 sm:p-5 rounded-2xl border border-slate-200 shadow-2xs">
        <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3">
          <div>
            <div className="flex items-center gap-2">
              <div className="w-8 h-8 rounded-xl bg-indigo-50 border border-indigo-200 text-indigo-700 flex items-center justify-center">
                <BarChart3 className="w-4 h-4" />
              </div>
              <h1 className="text-lg sm:text-xl font-bold text-slate-900">Trader Performance & Leak Analytics</h1>
            </div>
            <p className="text-xs text-slate-500 mt-1">
              Data-backed proof of your edge: Setup quality, discipline multiplier, and capital leaks.
            </p>
          </div>

          <button
            onClick={handleGenerateAiInsight}
            disabled={isAiLoading}
            className="px-3.5 py-2 rounded-xl text-xs font-bold bg-indigo-600 text-white hover:bg-indigo-700 transition-all flex items-center gap-1.5 shadow-xs"
          >
            <Sparkles className="w-4 h-4" />
            <span>{isAiLoading ? 'Analyzing Patterns...' : 'AI Performance Audit'}</span>
          </button>
        </div>
      </div>

      {/* AI Deep Insight (if generated) */}
      {aiInsight && (
        <div className="p-4 bg-gradient-to-r from-indigo-50/80 via-white to-slate-50 rounded-2xl border border-indigo-200 text-xs sm:text-sm text-slate-800 leading-relaxed shadow-2xs space-y-2">
          <div className="flex items-center gap-1.5 font-bold text-indigo-950">
            <Sparkles className="w-4 h-4 text-indigo-600" />
            <span>AI TRADER PERFORMANCE DEBRIEF</span>
          </div>
          <p>{aiInsight}</p>
        </div>
      )}

      {/* Key Metric Bento Grid */}
      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
        <div className="p-4 bg-white rounded-2xl border border-slate-200 shadow-2xs">
          <span className="text-xs text-slate-500 font-semibold block">Profit Factor</span>
          <div className="text-2xl sm:text-3xl font-extrabold font-mono text-slate-900 mt-1">
            {stats?.profitFactor ?? 0}
          </div>
          <span className="text-[11px] text-emerald-700 font-semibold mt-1 block">Gross Gains / Gross Losses</span>
        </div>

        <div className="p-4 bg-white rounded-2xl border border-slate-200 shadow-2xs">
          <span className="text-xs text-slate-500 font-semibold block">Mathematical Expectancy</span>
          <div className="text-2xl sm:text-3xl font-extrabold font-mono text-indigo-700 mt-1">
            +{(stats?.expectancy ?? (stats as any)?.expectancyR ?? 0)}R
          </div>
          <span className="text-[11px] text-slate-500 mt-1 block">Expected return per 1R risked</span>
        </div>

        <div className="p-4 bg-white rounded-2xl border border-slate-200 shadow-2xs">
          <span className="text-xs text-slate-500 font-semibold block">Rule Adherence Rate</span>
          <div className="text-2xl sm:text-3xl font-extrabold font-mono text-emerald-700 mt-1">
            {stats?.ruleAdherenceRate ?? 0}%
          </div>
          <span className="text-[11px] text-slate-500 mt-1 block">Constitution compliance</span>
        </div>

        <div className="p-4 bg-white rounded-2xl border border-slate-200 shadow-2xs">
          <span className="text-xs text-slate-500 font-semibold block">Max Recorded Drawdown</span>
          <div className="text-2xl sm:text-3xl font-extrabold font-mono text-rose-700 mt-1">
            ₹{(stats?.maxDrawdown ?? 0).toLocaleString('en-IN')}
          </div>
          <span className="text-[11px] text-slate-500 mt-1 block">Peak-to-trough capital dip</span>
        </div>
      </div>

      {/* CUMULATIVE PNL & EQUITY TRAJECTORY CHART */}
      <div className="p-4 sm:p-5 bg-white rounded-2xl border border-slate-200 shadow-2xs space-y-4">
        {/* Chart Header & Controls */}
        <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3 border-b border-slate-100 pb-3.5">
          <div>
            <div className="flex items-center gap-2">
              <div className="w-7 h-7 rounded-lg bg-emerald-50 border border-emerald-200 text-emerald-700 flex items-center justify-center">
                <TrendingUp className="w-4 h-4" />
              </div>
              <h2 className="text-sm sm:text-base font-bold text-slate-900">
                Cumulative P&L Curve & Equity Trajectory
              </h2>
            </div>
            <p className="text-xs text-slate-500 mt-0.5">
              Realized capital growth progression across your trade history ({cumulativeData.length - 1} logged trades)
            </p>
          </div>

          {/* Toggle between INR and R-Multiple */}
          <div className="flex items-center p-1 rounded-xl bg-slate-100 border border-slate-200 text-xs font-semibold self-stretch sm:self-auto justify-center">
            <button
              onClick={() => setChartMode('pnl')}
              className={`px-3 py-1.5 rounded-lg transition-all ${
                chartMode === 'pnl'
                  ? 'bg-white text-indigo-900 shadow-xs font-bold'
                  : 'text-slate-600 hover:text-slate-900'
              }`}
            >
              Cumulative P&L (₹)
            </button>
            <button
              onClick={() => setChartMode('rMultiple')}
              className={`px-3 py-1.5 rounded-lg transition-all ${
                chartMode === 'rMultiple'
                  ? 'bg-white text-indigo-900 shadow-xs font-bold'
                  : 'text-slate-600 hover:text-slate-900'
              }`}
            >
              Cumulative R (+R)
            </button>
          </div>
        </div>

        {/* Trajectory Highlights Bar */}
        <div className="grid grid-cols-2 sm:grid-cols-4 gap-2.5 text-xs">
          <div className="p-3 rounded-xl bg-slate-50 border border-slate-200/80">
            <span className="text-[11px] text-slate-500 font-medium block">Net Realized P&L</span>
            <div className={`text-base sm:text-lg font-mono font-extrabold mt-0.5 ${
              trajectoryStats.currentPnL >= 0 ? 'text-emerald-700' : 'text-rose-700'
            }`}>
              {trajectoryStats.currentPnL >= 0 ? '+' : ''}₹{trajectoryStats.currentPnL.toLocaleString('en-IN')}
            </div>
          </div>

          <div className="p-3 rounded-xl bg-slate-50 border border-slate-200/80">
            <span className="text-[11px] text-slate-500 font-medium block">Total Realized R</span>
            <div className="text-base sm:text-lg font-mono font-extrabold text-indigo-700 mt-0.5">
              {trajectoryStats.currentR >= 0 ? '+' : ''}{trajectoryStats.currentR}R
            </div>
          </div>

          <div className="p-3 rounded-xl bg-slate-50 border border-slate-200/80">
            <span className="text-[11px] text-slate-500 font-medium block">Peak Equity (High Watermark)</span>
            <div className="text-base sm:text-lg font-mono font-extrabold text-slate-900 mt-0.5">
              ₹{trajectoryStats.peakPnL.toLocaleString('en-IN')}
            </div>
          </div>

          <div className="p-3 rounded-xl bg-slate-50 border border-slate-200/80">
            <span className="text-[11px] text-slate-500 font-medium block">Current Dip from Peak</span>
            <div className={`text-base sm:text-lg font-mono font-extrabold mt-0.5 ${
              trajectoryStats.drawdownFromPeak === 0 ? 'text-emerald-700' : 'text-amber-700'
            }`}>
              {trajectoryStats.drawdownFromPeak === 0 ? '0 (At All-Time High)' : `-₹${trajectoryStats.drawdownFromPeak.toLocaleString('en-IN')}`}
            </div>
          </div>
        </div>

        {/* Recharts Visualization Container */}
        <div className="w-full h-64 sm:h-72 pt-2">
          <ResponsiveContainer width="100%" height="100%">
            <ComposedChart data={cumulativeData} margin={{ top: 10, right: 15, left: -10, bottom: 0 }}>
              <defs>
                <linearGradient id="pnlGradient" x1="0" y1="0" x2="0" y2="1">
                  <stop offset="5%" stopColor="#4f46e5" stopOpacity={0.18} />
                  <stop offset="95%" stopColor="#4f46e5" stopOpacity={0.0} />
                </linearGradient>
                <linearGradient id="rGradient" x1="0" y1="0" x2="0" y2="1">
                  <stop offset="5%" stopColor="#10b981" stopOpacity={0.18} />
                  <stop offset="95%" stopColor="#10b981" stopOpacity={0.0} />
                </linearGradient>
              </defs>
              <CartesianGrid strokeDasharray="3 3" stroke="#f1f5f9" vertical={false} />
              <XAxis
                dataKey="date"
                stroke="#94a3b8"
                fontSize={11}
                tickLine={false}
                axisLine={{ stroke: '#e2e8f0' }}
              />
              <YAxis
                stroke="#94a3b8"
                fontSize={11}
                tickLine={false}
                axisLine={{ stroke: '#e2e8f0' }}
                tickFormatter={(value) => {
                  if (chartMode === 'pnl') {
                    if (Math.abs(value) >= 1000) {
                      return `₹${(value / 1000).toFixed(value % 1000 === 0 ? 0 : 1)}k`;
                    }
                    return `₹${value}`;
                  }
                  return `${value > 0 ? '+' : ''}${value}R`;
                }}
              />
              <ReferenceLine y={0} stroke="#cbd5e1" strokeDasharray="3 3" />
              <Tooltip content={<CustomChartTooltip />} />
              <Area
                type="monotone"
                dataKey={chartMode === 'pnl' ? 'cumulativePnL' : 'cumulativeR'}
                fill={chartMode === 'pnl' ? 'url(#pnlGradient)' : 'url(#rGradient)'}
                stroke="none"
              />
              <Line
                type="monotone"
                dataKey={chartMode === 'pnl' ? 'cumulativePnL' : 'cumulativeR'}
                stroke={chartMode === 'pnl' ? '#4f46e5' : '#059669'}
                strokeWidth={2.5}
                dot={{
                  r: 3.5,
                  fill: chartMode === 'pnl' ? '#4f46e5' : '#059669',
                  strokeWidth: 2,
                  stroke: '#ffffff',
                }}
                activeDot={{
                  r: 6,
                  fill: chartMode === 'pnl' ? '#4f46e5' : '#059669',
                  stroke: '#ffffff',
                  strokeWidth: 2,
                }}
              />
            </ComposedChart>
          </ResponsiveContainer>
        </div>

        <div className="flex items-center justify-between text-[11px] text-slate-400 pt-2 border-t border-slate-100">
          <span className="flex items-center gap-1.5">
            <span className="w-2 h-2 rounded-full bg-indigo-600 inline-block" />
            Smooth trendline indicates steady edge compounding without volatile drawdown swings.
          </span>
          <span className="font-mono text-slate-500">
            {stats.totalTrades} Total Trades Logged
          </span>
        </div>
      </div>

      {/* DISCIPLINE MULTIPLIER: The Process vs Outcome Proof */}
      <div className="p-4 sm:p-5 bg-white rounded-2xl border border-slate-200 shadow-2xs space-y-3">
        <div className="flex items-center justify-between">
          <div className="flex items-center gap-2">
            <Shield className="w-4 h-4 text-emerald-600" />
            <h2 className="text-xs font-bold uppercase tracking-wider text-slate-800">
              The Discipline Multiplier (Edge Proof)
            </h2>
          </div>
          <span className="text-[11px] text-slate-500">Comparing Disciplined vs Impulsive Trades</span>
        </div>

        <div className="grid grid-cols-1 md:grid-cols-2 gap-4 pt-1">
          {/* Disciplined trades */}
          <div className="p-4 rounded-xl bg-emerald-50/70 border border-emerald-200 space-y-2">
            <div className="flex items-center justify-between">
              <span className="font-bold text-xs text-emerald-950">100% Constitution Compliant Trades</span>
              <span className="text-xs font-mono font-extrabold px-2 py-0.5 rounded bg-emerald-200 text-emerald-900">
                {stats.disciplinedTradesWinRate}% Win Rate
              </span>
            </div>
            <p className="text-xs text-emerald-900/90">
              When setups have 80+ Quality Score, 1:2+ R:R, and follow trading windows, your average trade yields <span className="font-bold font-mono">+{stats.avgWinR}R</span> with zero catastrophic blowups.
            </p>
          </div>

          {/* Undisciplined trades */}
          <div className="p-4 rounded-xl bg-rose-50/70 border border-rose-200 space-y-2">
            <div className="flex items-center justify-between">
              <span className="font-bold text-xs text-rose-950">Rule-Breaking / FOMO Trades</span>
              <span className="text-xs font-mono font-extrabold px-2 py-0.5 rounded bg-rose-200 text-rose-900">
                {stats.undisciplinedTradesWinRate}% Win Rate
              </span>
            </div>
            <p className="text-xs text-rose-900/90">
              Chasing midday moves or widening stop losses causes 78% of your total capital drawdown.
            </p>
          </div>
        </div>
      </div>

      {/* BEHAVIORAL LEAK DETECTOR (The Core Discipline System) */}
      <div className="p-4 sm:p-5 bg-white rounded-2xl border border-slate-200 shadow-2xs space-y-4">
        <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-2">
          <div className="flex items-center gap-2">
            <AlertTriangle className="w-5 h-5 text-amber-600" />
            <div>
              <h2 className="text-sm font-bold text-slate-900">Behavioral Leak Detector</h2>
              <p className="text-xs text-slate-500">Unconscious psychological leaks draining your trading capital</p>
            </div>
          </div>

          <span className="text-xs font-semibold px-2.5 py-1 rounded-full bg-amber-50 text-amber-800 border border-amber-200">
            {leaks.length} Identified Leaks
          </span>
        </div>

        <div className="space-y-3">
          {leaks.map((leak) => {
            const leakTitle = leak.title || (leak as any).name || 'Behavioral Leak';
            const drainText = (leak as any).estimatedLoss != null
              ? `Estimated Drain: ₹${Number((leak as any).estimatedLoss).toLocaleString('en-IN')}`
              : leak.evidence || leak.frequency || 'Pattern Detected';

            return (
              <div
                key={leak.id}
                className="p-4 rounded-xl border border-slate-200 hover:border-slate-300 transition-all bg-slate-50/60 space-y-2.5"
              >
                <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2">
                  <div className="flex items-center gap-2">
                    <span
                      className={`text-[10px] font-bold uppercase px-2 py-0.5 rounded-full ${
                        leak.severity?.toLowerCase() === 'high'
                          ? 'bg-rose-100 text-rose-800'
                          : 'bg-amber-100 text-amber-800'
                      }`}
                    >
                      {leak.severity || 'Medium'} Impact
                    </span>
                    <h3 className="font-bold text-xs sm:text-sm text-slate-900">{leakTitle}</h3>
                  </div>

                  <span className="text-xs font-mono font-bold text-rose-700 bg-rose-50 px-2 py-0.5 rounded border border-rose-200/80">
                    {drainText}
                  </span>
                </div>

                <p className="text-xs text-slate-600 leading-relaxed">
                  <span className="font-semibold text-slate-800">Behavioral Pattern: </span>
                  {leak.description}
                </p>

                <div className="p-2.5 rounded-lg bg-white border border-slate-200 text-xs flex flex-col sm:flex-row sm:items-center justify-between gap-2">
                  <div className="flex items-center gap-1.5 text-indigo-900 font-medium">
                    <Lightbulb className="w-4 h-4 text-indigo-600 shrink-0" />
                    <span>Fix: {leak.actionableFix}</span>
                  </div>

                  <button
                    onClick={() => onApplyLeakFixToConstitution(leak)}
                    className="px-3 py-1 rounded-lg text-xs font-bold bg-indigo-50 hover:bg-indigo-100 text-indigo-700 border border-indigo-200 transition-all shrink-0"
                  >
                    Enforce in Constitution
                  </button>
                </div>
              </div>
            );
          })}
        </div>
      </div>

export function toNumber(
  value: unknown,
  fallback = 0,
): number {
  if (
    typeof value === 'number' &&
    Number.isFinite(value)
  ) {
    return value;
  }

  if (typeof value === 'string') {
    const parsed = Number(value);

    if (Number.isFinite(parsed)) {
      return parsed;
    }
  }

  return fallback;
}

function toStringValue(
  value: unknown,
  fallback = '',
): string {
  if (
    typeof value === 'string' &&
    value.trim().length > 0
  ) {
    return value;
  }

  if (
    typeof value === 'number' ||
    typeof value === 'boolean'
  ) {
    return String(value);
  }

  return fallback;
}

function asRecord(
  value: unknown,
): Record<string, unknown> {
  if (
    value &&
    typeof value === 'object' &&
    !Array.isArray(value)
  ) {
    return value as Record<string, unknown>;
  }

  return {};
}

function getOhlc(
  quote: Record<string, unknown>,
): Record<string, unknown> {
  return asRecord(quote.ohlc);
}

function normalizeMarketQuote(
  value: unknown,
  fallbackInstrumentKey?: string,
): MarketQuote {
  const quote = asRecord(value);
  const ohlc = getOhlc(quote);

  const instrumentKey =
    toStringValue(
      quote.instrumentKey ??
        quote.instrument_key ??
        quote.instrument_token,
      fallbackInstrumentKey || '',
    );

  const symbol =
    toStringValue(
      quote.symbol ??
        quote.trading_symbol,
      '',
    );

  const lastPrice = toNumber(
    quote.last_price ??
      quote.lastPrice ??
      quote.ltp ??
      quote.LTP ??
      quote.price,
    0,
  );

  const previousClose = toNumber(
    quote.prev_close_price ??
      quote.previousClose ??
      quote.previous_close ??
      quote.cp ??
      ohlc.close,
    0,
  );

  const change = toNumber(
    quote.net_change ??
      quote.change,
    lastPrice - previousClose,
  );

  const changePercent = toNumber(
    quote.change_percent ??
      quote.changePercent ??
      quote.percentageChange,
    previousClose > 0
      ? (change / previousClose) * 100
      : 0,
  );

  const open = toNumber(
    quote.open ??
      ohlc.open,
    0,
  );

  const high = toNumber(
    quote.high ??
      ohlc.high,
    0,
  );

  const low = toNumber(
    quote.low ??
      ohlc.low,
    0,
  );

  const close = toNumber(
    quote.close ??
      ohlc.close,
    0,
  );

  const volume = toNumber(
    quote.volume ??
      ohlc.volume,
    0,
  );

  const averageTradedPrice = toNumber(
    quote.average_price ??
      quote.averagePrice ??
      quote.atp,
    0,
  );

  const timestamp =
    toStringValue(
      quote.timestamp ??
        quote.updatedAt,
      '',
    );

  return {
    ...quote,

    instrumentKey:
      instrumentKey || fallbackInstrumentKey,

    symbol,

    ltp: lastPrice,
    lastPrice,
    price: lastPrice,

    change,
    changePercent,
    percentageChange: changePercent,

    open,
    high,
    low,
    close,
    previousClose,

    volume,
    averageTradedPrice,

    timestamp: timestamp || undefined,
    updatedAt: timestamp || undefined,
  };
}

function extractQuoteRecords(
  response: unknown,
): MarketQuote[] {
  let payload: unknown = response;

  if (
    payload &&
    typeof payload === 'object' &&
    'data' in payload
  ) {
    const data =
      (payload as {
        data?: unknown;
      }).data;

    if (data !== undefined) {
      payload = data;
    }
  }

  if (Array.isArray(payload)) {
    return payload.map((item) =>
      normalizeMarketQuote(item),
    );
  }

  const record = asRecord(payload);

  const directArrays = [
    record.market,
    record.quotes,
    record.items,
  ];

  for (const candidate of directArrays) {
    if (Array.isArray(candidate)) {
      return candidate.map((item) =>
        normalizeMarketQuote(item),
      );
    }
  }

  return Object.entries(record)
    .filter(([, value]) =>
      Boolean(
        value &&
          typeof value === 'object' &&
          !Array.isArray(value),
      ),
    )
    .map(([key, value]) => {
      const quote = asRecord(value);

      const instrumentKey =
        toStringValue(
          quote.instrumentKey ??
            quote.instrument_key ??
            quote.instrument_token,
          key.replace(':', '|'),
        );

      return normalizeMarketQuote(
        quote,
        instrumentKey,
      );
    });
}

function extractSingleQuote(
  response: unknown,
  instrumentKey: string,
): MarketQuote | null {
  let payload: unknown = response;

  if (
    payload &&
    typeof payload === 'object' &&
    'data' in payload
  ) {
    const data =
      (payload as {
        data?: unknown;
      }).data;

    if (data !== undefined) {
      payload = data;
    }
  }

  if (Array.isArray(payload)) {
    if (payload.length === 0) {
      return null;
    }

    return normalizeMarketQuote(
      payload[0],
      instrumentKey,
    );
  }

  const record = asRecord(payload);

  const possibleKeys = [
    instrumentKey,
    instrumentKey.replace('|', ':'),
  ];

  for (const key of possibleKeys) {
    const candidate = record[key];

    if (
      candidate &&
      typeof candidate === 'object' &&
      !Array.isArray(candidate)
    ) {
      return normalizeMarketQuote(
        candidate,
        instrumentKey,
      );
    }
  }

  const directQuoteFields = [
    'last_price',
    'lastPrice',
    'ltp',
    'LTP',
    'price',
  ];

  if (
    directQuoteFields.some(
      (key) => key in record,
    )
  ) {
    return normalizeMarketQuote(
      record,
      instrumentKey,
    );
  }

  const entries = Object.entries(record);

  for (const [, value] of entries) {
    if (
      !value ||
      typeof value !== 'object' ||
      Array.isArray(value)
    ) {
      continue;
    }

    const candidate =
      value as Record<string, unknown>;

    const candidateInstrument =
      candidate.instrument_token ??
      candidate.instrumentKey ??
      candidate.instrument_key;

    if (
      candidateInstrument === instrumentKey
    ) {
      return normalizeMarketQuote(
        candidate,
        instrumentKey,
      );
    }
  }

  if (entries.length === 1) {
    const value = entries[0][1];

    if (
      value &&
      typeof value === 'object' &&
      !Array.isArray(value)
    ) {
      return normalizeMarketQuote(
        value,
        instrumentKey,
      );
    }
  }

  return null;
}

async function request<T>(
  path: string,
  options: RequestInit = {},
): Promise<T> {
  let response: Response;

  try {
    response = await fetch(
      `${API_PREFIX}${path}`,
      {
        ...options,
        headers: {
          Accept: 'application/json',

          ...(options.body
            ? {
                'Content-Type':
                  'application/json',
              }
            : {}),

          ...getAuthHeaders(),

          ...(options.headers || {}),
        },

        credentials: 'same-origin',
      },
    );
  } catch {
    throw new QuanttoraApiError(
      'Quanttora backend is not reachable. Start the Dart backend on port 8080.',
      0,
    );
  }

  const contentType =
    response.headers.get(
      'content-type',
    ) || '';

  let payload: unknown;

  if (
    contentType.includes(
      'application/json',
    )
  ) {
    payload = await response.json();
  } else {
    const text =
      await response.text();

    payload = text || null;
  }

  if (!response.ok) {
    const errorPayload =
      payload &&
      typeof payload === 'object'
        ? (payload as ApiResponse<unknown>)
        : null;

    const message =
      errorPayload?.error?.message ||
      errorPayload?.message ||
      (
        response.status === 401
          ? 'Your Quanttora session has expired. Please sign in again.'
          : `Quanttora backend request failed (${response.status})`
      );

    if (response.status === 401) {
      clearSession();
    }

    throw new QuanttoraApiError(
      message,
      response.status,
      errorPayload?.error?.code,
    );
  }

  return payload as T;
}

function unwrap<T>(
  response: T | ApiResponse<T>,
): T {
  if (
    response &&
    typeof response === 'object' &&
    'data' in response
  ) {
    const data =
      (response as ApiResponse<T>)
        .data;

    if (data !== undefined) {
      return data;
    }
  }

  return response as T;
}

function installAuthenticatedFetch(): () => void {
  const originalFetch =
    window.fetch.bind(window);

  const authenticatedFetch = async (
    input: RequestInfo | URL,
    init?: RequestInit,
  ): Promise<Response> => {
    let url = '';

    if (typeof input === 'string') {
      url = input;
    } else if (
      input instanceof URL
    ) {
      url = input.toString();
    } else {
      url = input.url;
    }

    const isQuanttoraApiRequest =
      url.startsWith('/api/') ||
      url.startsWith(
        window.location.origin + '/api/',
      );

    const isLoginRequest =
      url.includes(
        '/api/auth/login',
      ) ||
      url.includes(
        '/api/auth/register',
      );

    const isUpstoxLoginRequest =
      url.includes(
        '/api/auth/upstox/login',
      );

    if (
      !isQuanttoraApiRequest ||
      isLoginRequest
    ) {
      return originalFetch(
        input,
        init,
      );
    }

    const token =
      getAuthToken();

    if (!token) {
      return originalFetch(
        input,
        init,
      );
    }

    const headers =
      new Headers(
        init?.headers,
      );

    if (
      !headers.has(
        'Authorization',
      )
    ) {
      headers.set(
        'Authorization',
        `Bearer ${token}`,
      );
    }

    if (
      isUpstoxLoginRequest &&
      !headers.has(
        'Accept',
      )
    ) {
      headers.set(
        'Accept',
        'application/json',
      );
    }

    return originalFetch(
      input,
      {
        ...init,
        headers,
      },
    );
  };

  window.fetch =
    authenticatedFetch;

  return () => {
    window.fetch =
      originalFetch;
  };
}

export default function App() {
  const [
    authSession,
    setAuthSession,
  ] =
    useState<QuanttoraSession | null>(
      () => getStoredSession(),
    );

  useEffect(() => {
    if (!authSession) {
      return;
    }

    const restoreSession =
      getStoredSession();

    if (!restoreSession) {
      setAuthSession(null);
      return;
    }

    const cleanup =
      installAuthenticatedFetch();

    return cleanup;
  }, [authSession]);

  const handleAuthenticated = (
    session: QuanttoraSession,
  ) => {
    setAuthSession(
      session,
    );
  };

  const handleLogout = () => {
    clearSession();
    setAuthSession(null);
  };

  /*
   * Authentication is intentionally handled
   * before the trading application mounts.
   *
   * This prevents protected market/broker requests
   * from being made without a Quanttora session.
   */
  if (!authSession) {
    return (
      <LoginView
        onAuthenticated={
          handleAuthenticated
        }
      />
    );
  }

  return (
    <AuthenticatedDashboard
      onLogout={
        handleLogout
      }
    />
  );
}

interface AuthenticatedDashboardProps {
  onLogout: () => void;
}

function AuthenticatedDashboard({
  onLogout,
}: AuthenticatedDashboardProps) {
  useEffect(() => {
    /*
     * DashboardApp currently owns the profile modal
     * and logout callback internally.
     *
     * The global authenticated session remains valid
     * until the user explicitly signs out or the backend
     * rejects the session.
     *
     * This effect intentionally does not duplicate the
     * dashboard's existing state.
     */
    return undefined;
  }, []);

  return (
    <DashboardSessionBoundary
      onLogout={
        onLogout
      }
    />
  );
}

interface DashboardSessionBoundaryProps {
  onLogout: () => void;
}

function DashboardSessionBoundary({
  onLogout,
}: DashboardSessionBoundaryProps) {
  /*
   * Keep the existing DashboardApp untouched.
   *
   * The logout callback is exposed through the
   * browser event below. The dashboard profile
   * action can dispatch this event when needed.
   */
  useEffect(() => {
    const handleSessionLogout =
      () => {
        onLogout();
      };

    window.addEventListener(
      'quanttora:logout',
      handleSessionLogout,
    );

    return () => {
      window.removeEventListener(
        'quanttora:logout',
        handleSessionLogout,
      );
    };
  }, [onLogout]);

  return (
    <DashboardApp />
  );
}

function findArray(
  source: Record<string, unknown>,
  ...keys: string[]

function pickNumber(
  source: Record<string, unknown>,

      return value;

  return fallback;

    recommendationSummary:
      pickString(
        raw,
        connected
          ? 'Live market data is available. Assess your setup before execution.'
          : 'Live market data is unavailable. Do not treat displayed market values as current.',
        'recommendationSummary',
        'summary'
      ),

    keyLevels: {
      niftySupport:
        Array.isArray(
          keyLevels.niftySupport
        )
          ? keyLevels.niftySupport
              .map((value) =>
                toNumber(value)
              )
              .filter(Boolean)
          : [],

      niftyResistance:
        Array.isArray(
          keyLevels.niftyResistance
        )
          ? keyLevels.niftyResistance
              .map((value) =>
                toNumber(value)
              )
              .filter(Boolean)
          : [],

      bankNiftySupport:
        Array.isArray(
          keyLevels.bankNiftySupport
        )
          ? keyLevels.bankNiftySupport
              .map((value) =>
                toNumber(value)
              )
              .filter(Boolean)
          : [],

      bankNiftyResistance:
        Array.isArray(
          keyLevels.bankNiftyResistance
        )
          ? keyLevels.bankNiftyResistance
              .map((value) =>
                toNumber(value)
              )
              .filter(Boolean)
          : [],
    },

    importantEvents,

    preMarketSummary:
      pickString(
        raw,
        connected
          ? 'Live market feed connected.'
          : 'Market feed is offline.',
        'preMarketSummary',
        'marketSummary'
      ),

    globalContext: pickString(
      raw,
      '',
      'globalContext'
    ),
  };
}

  // ============================================================

    PerformanceStats = useMemo(() => {
    if (trades.length === 0) {
      return initialPerformanceStats;

    const totalPnL =
      trades.reduce(
        (acc, trade) =>

      Math.abs(
        losingTrades.reduce(

      totalLosses > 0

              winningTrades.length

                losingTrades.reduce(
                  (acc, trade) =>
                    acc +
                    trade.realizedR,
                  0
                )
              ) /
              losingTrades.length
            ).toFixed(2)
          )
        : 1.0;

    const winProb =
      winRate / 100;

    const lossProb =
      (100 - winRate) /
      100;

    const expectancy =
      Number(
        (
          winProb * avgWinR -
          lossProb * avgLossR
        ).toFixed(2)
      );

    const compliantTrades =
      trades.filter(
        (trade) =>
          trade.ruleCompliance
      );

    const nonCompliantTrades =
      trades.filter(
        (trade) =>
          !trade.ruleCompliance
      );

    const disciplinedWinRate =
      compliantTrades.length > 0
        ? Number(
            (
              (compliantTrades.filter(
                (trade) =>
                  trade.pnl > 0
              ).length /
                compliantTrades.length) *
              100
            ).toFixed(1)
          )
        : 80;

    const undisciplinedWinRate =
      nonCompliantTrades.length > 0
        ? Number(
            (
              (nonCompliantTrades.filter(
                (trade) =>
                  trade.pnl > 0
              ).length /
                nonCompliantTrades.length) *
              100
            ).toFixed(1)
          )
        : 25;

      Number(
        (
          (compliantTrades.length /
            trades.length) *
          100
        ).toFixed(1)
      );

    return {
      totalPnL,
      winRate,
      totalTrades:
        trades.length,
      winningTrades:
        winningTrades.length,
      losingTrades:
        losingTrades.length,

  };

  useEffect(() => {
    let cancelled = false;

    const wait = (milliseconds: number) =>
      new Promise<void>((resolve) => {
        window.setTimeout(resolve, milliseconds);
      });

    const initialiseBrokerConnection = async () => {
      const params =
        new URLSearchParams(

      const brokerAuth =
        params.get('broker_auth');

      const returnedFromOAuth =
        brokerAuth === 'success' ||
        brokerAuth === 'error';

      if (returnedFromOAuth) {

          document.title,

        // The Upstox callback can finish just before the browser remounts
        // this page. During that short window the first broker/status call
        // may legitimately return 401 or report the previous state. Retry
        // the real backend status instead of marking the broker disconnected
        // permanently. This is especially important immediately after OAuth.
        const maxAttempts = returnedFromOAuth
          ? 12
          : 4;

        let brokerConnected = false;

        for (
          let attempt = 1;
          attempt <= maxAttempts;
          attempt += 1
        ) {
          if (cancelled) {
            return;
          }

          brokerConnected =
            await syncUpstoxConnection();

          if (brokerConnected) {
            break;
          }

          if (attempt < maxAttempts) {
            await wait(1000);
          }
        }

        if (
          cancelled ||
          !brokerConnected
        ) {
          return;
        }

        const feedStatus =
          await getBrokerFeedStatus();

          cancelled ||
          feedStatus.connected

          return;

          'upstox',
        );

          );

      }

    return () => {
      cancelled = true;

  // ============================================================

  return (

        }

        onToggleStandDown={
          handleToggleStandDown
        }
      />

      <StandDownBanner
        constitution={
          constitution
        }

        onDeactivate={
          handleDeactivateStandDown
        }

        onOpenConstitution={() =>
          setIsConstitutionModalOpen(
            true
          )
        }
      />

      <main className="flex-1 max-w-7xl w-full mx-auto px-2 sm:px-4 pt-3 sm:pt-4">

        {activeTab === 'home' && (
          <HomeDashboard
            indices={
              indices
            }

            breadth={
              breadth
            }

            sectors={
              sectors
            }

            marketIntel={
              marketIntel
            }

            constitution={
              constitution
            }

            discoveries={
              discoveries
            }

            stats={
              computedStats
            }

            knowledgeScore={
              knowledgeScore
            }

            marketFeedConnected={
              marketFeedConnected
            }

            marketFeedLoading={
              marketFeedLoading
            }

            marketFeedError={
              marketFeedError
            }

            marketLastUpdated={
              marketLastUpdated
            }

            onNavigate={(
              tab
            ) =>
              setActiveTab(
                tab
              )
            }

            onSelectDiscoveryForDecision={
              handleSelectDiscoveryForDecision
            }

            onOpenConstitution={() =>
              setIsConstitutionModalOpen(
                true
              )
            }

            onToggleStandDown={
              handleToggleStandDown
            }
          />
        )}

        {activeTab ===
          'discover' && (
          <DiscoverView
            discoveries={
              discoveries
            }

            onSelectForDecision={
              handleSelectDiscoveryForDecision
            }

            onNavigate={(
              tab
            ) =>
              setActiveTab(
                tab
              )
            }
          />
        )}

        {activeTab ===
          'decision' && (
          <DecisionEngineView
            constitution={
              constitution
            }

            initialStock={
              selectedStockForDecision
            }

            onLogTradeToJournal={
              handleLogTrade
            }

            onNavigate={(
              tab
            ) =>
              setActiveTab(
                tab
              )
            }

            onOpenConstitution={() =>
              setIsConstitutionModalOpen(
                true
              )
            }
          />
        )}

        {activeTab ===
          'riskguard' && (
          <RiskGuardView
            onNavigate={(tab) =>
              setActiveTab(
                tab
              )
            }
          />
        )}

        {activeTab ===
          'journal' && (
          <JournalView
            trades={
              trades
            }

            stats={
              computedStats
            }

            constitution={
              constitution
            }

            onAddTrade={
              handleLogTrade
            }

            onNavigateToAnalytics={() =>
              setActiveTab(
                'analytics'
              )
            }
          />
        )}

        {activeTab ===
          'analytics' && (
          <AnalyticsView
            stats={
              computedStats
            }

            trades={
              trades
            }

            leaks={
              leaks
            }

            constitution={
              constitution
            }

            onApplyLeakFixToConstitution={
              handleApplyLeakFixToConstitution
            }

            onNavigate={(
              tab
            ) =>
              setActiveTab(
                tab
              )
            }
          />
        )}

        {activeTab ===
          'academy' && (
          <AcademyView
            lessons={
              lessons
            }

            knowledgeScore={
              knowledgeScore
            }

            onCompleteLesson={
              handleCompleteLesson
            }

            onOpenPractice={() =>
              setActiveTab(
                'practice'
              )
            }
          />
        )}

        {activeTab ===
          'practice' && (
          <PracticeSimulatorView
            scenarios={
              initialPracticeScenarios
            }

            onEarnPracticePoints={
              handleEarnPracticePoints
            }

            onNavigateBack={() =>
              setActiveTab(
                'academy'
              )
            }
          />
        )}

        {activeTab ===
          'brokers' && (
          <BrokersView
            brokers={
              brokers
            }

            onToggleConnect={
              handleToggleBroker
            }

            onClose={() =>
              setActiveTab(
                'home'
              )
            }
          />
        )}

      </main>

      <BottomNav
        activeTab={
          activeTab
        }

        onSelectTab={(
          tab
        ) =>
          setActiveTab(
            tab
          )
        }

        decisionBadgeCount={
          selectedStockForDecision
            ? 1
            : 0
        }
      />

      <ConstitutionModal
        isOpen={
          isConstitutionModalOpen
        }

        onClose={() =>
          setIsConstitutionModalOpen(
            false
          )
        }

        constitution={
          constitution
        }

        onSaveConstitution={
          handleSaveConstitution
        }
      />

      <TraderProfileModal
        isOpen={
          isProfileModalOpen
        }

        onClose={() =>
          setIsProfileModalOpen(
            false
          )
        }

        knowledgeScore={
          knowledgeScore
        }

        constitution={
          constitution
        }

        onResetData={
          handleResetData
        }
      />
    </div>
  );
}
